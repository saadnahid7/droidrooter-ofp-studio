cmake_minimum_required(VERSION 3.20)
project(ofp_studio VERSION 1.1.0 LANGUAGES C RC)
set(CMAKE_C_STANDARD 11)
if(NOT WIN32)
  message(FATAL_ERROR "OFP Studio uses Windows system APIs. Build on Windows.")
endif()
add_library(ofp_core STATIC src/archive.c src/crypto.c src/mtk.c src/qc.c src/extract.c src/scatter.c src/output.c)
target_include_directories(ofp_core PUBLIC src)
target_compile_definitions(ofp_core PUBLIC UNICODE _UNICODE _WIN32_WINNT=0x0A00)
target_link_libraries(ofp_core PUBLIC bcrypt xmllite shlwapi ole32 uuid)
if(MINGW)
  target_compile_options(ofp_core PRIVATE -Wall -Wextra -Wno-misleading-indentation)
  add_compile_options(-Wall -Wextra -Wno-misleading-indentation)
  add_link_options(-static -static-libgcc -municode)
endif()
add_executable(ofp-test tests/driver.c)
target_link_libraries(ofp-test PRIVATE ofp_core)
find_program(OFP_CARGO NAMES cargo HINTS "$ENV{OFP_CARGO_DIR}" REQUIRED)
set(MAGISK_APK "${CMAKE_CURRENT_SOURCE_DIR}/.local/magisk/Magisk-v31.1.apk" CACHE FILEPATH "Verified official Magisk 31 APK, embedded as data")
if(NOT EXISTS "${MAGISK_APK}")
  message(FATAL_ERROR "Provide MAGISK_APK pointing to official Magisk-v31.1.apk (see build instructions).")
endif()
file(SHA256 "${MAGISK_APK}" MAGISK_HASH)
if(NOT MAGISK_HASH STREQUAL "89cefac0a318afa39fb3880c45c87b036c05bc1ebf02a7a715b7f8d7c26ac6e7")
  message(FATAL_ERROR "Official Magisk 31 APK digest mismatch")
endif()
file(TO_CMAKE_PATH "${MAGISK_APK}" MAGISK_APK_RESOURCE)
configure_file(assets/magisk.rc.in "${CMAKE_BINARY_DIR}/magisk.rc" @ONLY)
set(MAGISK_LIB "${CMAKE_BINARY_DIR}/native/release/libofp_magisk_engine.a")
file(GLOB_RECURSE MAGISK_RS CONFIGURE_DEPENDS native/magisk-engine/src/*.rs)
add_custom_command(OUTPUT "${MAGISK_LIB}"
  COMMAND ${CMAKE_COMMAND} -E env "CARGO_TARGET_DIR=${CMAKE_BINARY_DIR}/native" "RUSTFLAGS=-C linker=rust-lld -C linker-flavor=ld.lld -C link-self-contained=yes"
          "${OFP_CARGO}" +stable-x86_64-pc-windows-gnu build --release --locked --manifest-path "${CMAKE_CURRENT_SOURCE_DIR}/native/magisk-engine/Cargo.toml"
  DEPENDS ${MAGISK_RS} native/magisk-engine/Cargo.toml native/magisk-engine/Cargo.lock
  VERBATIM)
add_custom_target(magisk_build DEPENDS "${MAGISK_LIB}")
add_library(magisk_engine STATIC IMPORTED GLOBAL)
set_target_properties(magisk_engine PROPERTIES IMPORTED_LOCATION "${MAGISK_LIB}")
add_dependencies(magisk_engine magisk_build)
add_library(magisk_io STATIC src/magisk_io.c)
target_link_libraries(magisk_io PUBLIC ofp_core magisk_engine ntdll userenv ws2_32 crypt32 secur32 advapi32 bcrypt mswsock synchronization)
add_executable(magisk-test tests/magisk_driver.c "${CMAKE_BINARY_DIR}/magisk.rc")
target_link_libraries(magisk-test PRIVATE magisk_io)
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/src/gui.c")
  add_executable(OFPStudio WIN32 src/gui.c src/magisk_gui.c assets/app.rc "${CMAKE_BINARY_DIR}/magisk.rc")
  target_link_libraries(OFPStudio PRIVATE ofp_core magisk_io comctl32 comdlg32 shell32 gdi32 uxtheme dwmapi)
  if(MINGW)
    target_link_options(OFPStudio PRIVATE -s)
  endif()
endif()
